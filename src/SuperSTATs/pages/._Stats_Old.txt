import SuperSTATs.Main as stats
import streamlit as st
import altair as alt
import pandas as pd
import tbapy

st.set_page_config(
    page_title="[STATS] SN Stats",
    page_icon="🤖",
    initial_sidebar_state="expanded",
)


point_values = {
    "Auton Mobility": 3,
    "Auton Bottom": 3,
    "Auton Middle": 4,
    "Auton Top": 6,
    "Auton Docked NOT engaged": 8,
    "Auton Docked ENGAGED": 12,
    "Bottom Row": 2,
    "Middle Row": 3,
    "Top Row": 5,
    "Parked": 2,
    "Docked NOT engaged": 6,
    "Docked ENGAGED": 10,
}


def add_cols(org_df, reg, filter):
    org_df = org_df.drop(columns=org_df.filter(regex=".1").columns)
    org_df[filter] = org_df.filter(regex=reg).sum(axis=1)
    return org_df[filter]


def custom_bar_chart(data):
    chart = (
        alt.Chart(data)
        .mark_bar()
        .encode(
            x=alt.X("value:Q", title=" "),
            y=alt.Y(
                "Team:N",
                sort=alt.EncodingSortField(field="rank", order="ascending"),
                title="",
            ),
            color=alt.Color("variable:N", title=" "),
            tooltip=["Team:N", "value:Q", "variable:N"],
        )
        .properties(width=700, height=1500)
    )
    return chart


def value_convert(df, reg, value):
    for col in df.filter(regex=reg):
        df[f"{col} conv"] = df[col] * value


def config_chart(column, agg):
    df = stats.download_sheets_raw("1318549794")

    for point, value in point_values.items():
        value_convert(df, f"^{point}", value)

    df["Autonomous"] = add_cols(
        df,
        r"^Auton Bottom conv|^Auton Middle conv|^Auton Top conv|^Auton Mobility conv|^Auton Docked NOT engaged conv|^Auton Docked ENGAGED conv",
        "autonomous",
    )
    df["Teleop"] = add_cols(
        df, r"^Bottom Row conv|^Middle Row conv|^Top Row conv", "teleop"
    )
    df["Endgame"] = add_cols(
        df, r"^Docked ENGAGED conv|^Docked NOT engaged conv|^Parked conv", "endgame"
    )

    calc_df = df.groupby("Team")[column].agg(agg)
    calc_df["rank"] = calc_df.sum(axis=1).rank(ascending=False).astype(int)
    calc_df.reset_index(inplace=True)
    calc_row = pd.melt(
        calc_df,
        id_vars=["Team", "rank"],
        value_vars=column,
        var_name="variable",
        value_name="value",
    )
    return calc_row


def main():
    stats.set_font()

    st.header("NURD Stats")

    aggType = st.selectbox("Aggregation Type", ("Mean", "Median", "Sum")).lower()
    groupFilters = st.checkbox("Group Filters", True)

    if groupFilters:
        filter_cols = ["Autonomous", "Teleop", "Endgame"]
    else:
        filter_cols = [
            "Auton Mobility",
            "Auton Top",
            "Auton Middle",
            "Auton Bottom",
            "Auton Docked NOT engaged",
            "Auton Docked ENGAGED",
            "Top Row",
            "Middle Row",
            "Bottom Row",
            "Parked",
            "Docked NOT engaged",
            "Docked ENGAGED",
        ]

    selected_filters = st.multiselect("Filters:", filter_cols, default=filter_cols)

    if selected_filters:
        st.altair_chart(custom_bar_chart(config_chart(selected_filters, aggType)))
    else:
        st.error("Please select at least one filter")


if __name__ == "__main__":
    main()
